export class WebsocketClient {
  private reconnectInterval: number = 30_000;
  private reconnectTimer?: NodeJS.Timeout;
  
  // Логіка перепідключення з перевіркою інтервалу
  private _setReconnectInterval = () => {
    this.reconnectTimer = setInterval(() => {
      const readyState = this.socket?.getReadyState();
      if (!this.socket || readyState === WebSocket.CLOSED) {
        this._clearSocket();
        this._initializeSocket();
      }
    }, this.reconnectInterval);
  };
  
  // Паттерн handler для кількох підписників
  private handleCandleMessage({ id, contents }: CandleMessage) {

    if (subscriptionItem) {

      subscriptionItem.lastBar = bar;
      // Трансляція всім підписникам
      Object.values(subscriptionItem.handlers).forEach(
        (handler: HandlerType) => {
          handler.callback(bar);
        }
      );
    }
  }
}

  persist(
    (set, get) => ({
      setMarkets: (markets) => {
        const { selectedMarket } = get();
        // Селективні оновлення - оновлюємо тільки якщо змінилося

export const useDebounce = <T>(value: T, delay: number): T => {
  const [debouncedValue, setDebouncedValue] = useState<T>(value);
  
  useEffect(() => {
    const handler = setTimeout(() => setDebouncedValue(value), delay);
    return () => clearTimeout(handler);
  }, [value, delay]);
  
  return debouncedValue;
};

1. Паттерн батчингу WebSocket
class BatchedWebSocketClient {
  private updateQueue: MarketUpdate[] = [];
  private batchInterval = 16; // ~60fps
  private rafId: number | null = null;
  
  handleUpdate(update: MarketUpdate) {
    this.updateQueue.push(update);
    
    if (!this.rafId) {
      this.rafId = requestAnimationFrame(() => {
        this.flushUpdates();
        this.rafId = null;
      });
    }
  }
  
  private flushUpdates() {
    if (this.updateQueue.length === 0) return;
    
    // Об'єднання оновлень за символом
    const merged = this.updateQueue.reduce((acc, update) => {
      acc[update.symbol] = update;
      return acc;
    }, {} as Record<string, MarketUpdate>);
    
    // Одне оновлення стану
    this.onBatchUpdate(Object.values(merged));
    this.updateQueue = [];
  }
}

3. WebSocket з експоненційним backoff
class WebSocketClient {
  private reconnectAttempts = 0;
  private maxReconnectAttempts = 10;
  private baseDelay = 1000;
  
  private reconnect() {
    if (this.reconnectAttempts >= this.maxReconnectAttempts) {
      console.error('Max reconnection attempts reached');
      return;
    }
    
    const delay = Math.min(
      this.baseDelay * Math.pow(2, this.reconnectAttempts),
      30000 // Макс 30 секунд
    );
    
    setTimeout(() => {
      this.reconnectAttempts++;
      this.connect();
    }, delay);
  }
  
  private onOpen() {
    this.reconnectAttempts = 0; // Скидання при успішному з'єднанні
  }
}


ПІДГОТОВКА ДО PERFORMANCE PROFILING

Точки Chrome DevTools Performance Trace:

1. Аналіз частоти оновлень:
   - Записати performance trace
   - Визначити частоту оновлень (має бути ~16мс для 60fps)
   - Знайти вузькі місця (довгі задачі >50мс)

2. Вузькі місця для пошуку:
   - Layout thrashing (примусові reflows)
   - Надмірні re-render
   - Витоки пам'яті (зростаючий heap)
   - Довге виконання JavaScript

3. Виправлення для демонстрації:
   - До: Кожне WebSocket повідомлення викликає re-render
   - Після: Батчовані оновлення з requestAnimationFrame
   - До: Повний re-render таблиці при оновленні одного рядка
   - Після: Селективні оновлення з React.memo

Приклад пояснення Performance Trace:

"Ось performance trace з нашої торгової платформи:

1. Частота оновлень: Ми отримуємо ~60 WebSocket повідомлень на секунду
2. Початкова проблема: Кожне повідомлення викликало повний re-render дерева компонентів (~45мс)
3. Виявлене вузьке місце: 
   - Компонент MarketTable рендерить всі рядки
   - Оновлення store Zustand викликають непотрібні підписки
4. Застосовані виправлення:
   - Реалізовано батчинг: Оновлення збираються протягом 16мс вікна, потім один render
   - Додано React.memo до компонентів MarketRow
   - Використано селектори Zustand для запобігання непотрібним підпискам
5. Результат: Зменшено час рендерингу до ~3мс на батч, підтримуючи 60fps"
